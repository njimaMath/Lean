Assume that $d\geq 3$. In this section, we prove the following lemma. 
\begin{lem}
  \label{lem:2d-2}
  Let $x_n\in \partial D_n:=\{x\in\Z^d\colon \|x\|_1=n\}$ and $x_{n+1}\in \partial D_{n+1}$. 
 Assume that for some $r,k\in [d]$
 \begin{align}
\label{eq:OppositeSign}
    x_n(r)x_{n+1}(r)<0
\end{align}
 and
\begin{align*}
  x_n&=(x_n(1),x_n(2),\ldots,x_n(k),0,\ldots,0)
\end{align*} Then for any $\delta\le 1/8d$, there exist 
  \begin{align}
    &\text{$2d-|\{i\colon x_n(i)\neq 0\}|-1$ paths starting from $x_n$ when $x_n\notin \{\pm n\bfe_i\}_{i\in [d]}$,}\label{eq:xn_neq_ne}\\
    &\text{$2d-3$ paths starting from $x_n$ when $x_n=\pm n\bfe_i$ with some $i\in [d]$,}\label{eq:xn_eq_ne}\\
    &\text{$|\{j\colon x_{n+1}(j)\neq 0\}|-1$ paths starting from $x_{n+1}$ when $x_{n+1}\not\in \{\pm (n+1)\bfe_j\}_{j\in [d]}$,}\label{eq:xn+1_neq_n+1e}\\
    &\text{a path starting from $x_{n+1}$ when $x_{n+1}=\pm n\bfe_i $ with some $i\in [d]$,}\label{eq:xn+1_eq_n+1e}
\end{align}
that are edge-disjoint, stay in $\partial D_n\cup \partial D_{n+1}$, have length between $\lfloor \delta^3(n+1)\rfloor$\footnote{\SN $\delta^2 (n+1)?$} and $\lfloor 2d\delta^2(n+1)\rfloor$, and have endpoints at least $\delta^3(n+1)$ away from the union of all the other paths. 
\end{lem}
Note that for all sufficiently large $n$, 
\aln{\label{eq: xn coordinate large}
\text{$x_n$ has a coordinate that is larger than $n/d > 4 \delta^2 (n+1)$,}
}
which we use in this proof without explicit mention. 
\begin{proof}
\begin{lem}
\label{lem:path-separation}
Suppose that two paths $\gamma$ and $\gamma'$ have length at least $\delta^2 (n+1)$ and that there is $i\in[d]$ such that the following three conditions hold: 
\begin{align}
\label{eq:ps1}
    \text{the $i$-th coordinate of $\gamma_{\lfloor \delta^3 (n+1)\rfloor}$ is larger than that of $\gamma'_{\lfloor \delta^3 (n+1)\rfloor}$}
\end{align}
and after the first $\lfloor \delta^3 (n+1)\rfloor$ steps, 
\begin{align}
\label{eq:ps2}
    &\text{the $i$-th coordinate of $\gamma$ strictly  increases in every $2d$ steps, and}\\
\label{eq:ps3}
    &\text{the $i$-th coordinate of $\gamma'$ does not increase. }
\end{align}

Then the endpoint of $\gamma$ is at least $\delta^3(n+1)$ away from the whole path $\gamma'$. 
\end{lem}
\begin{proof}
By the assumption, it follows that the endpoint of $\gamma$ is at least 
\begin{equation*}
\frac{1}{2d} (\delta^2-\delta^3)(n+1) \ge 2\delta^3(n+1),\qquad \text{for } \delta \le (6d+1)^{-1}
\end{equation*}
away from $\{\gamma'_j\}_{j\ge \lfloor \delta^3(n+1)\rfloor}$ in the $i$-th coordinate. Since $\{\gamma'_j\}_{j\le \lfloor \delta^3(n+1)\rfloor}$ is contained in the $\delta^3(n+1)$ neighborhood of $\{\gamma'_j\}_{j\ge \lfloor \delta^3(n+1)\rfloor}$, it is also at least $\delta^3(n+1)$ away from the endpoint of $\gamma$. 
\end{proof}
In what follows, we will construct paths starting from $x_n$ and $x_{n+1}$ satisfying the conditions in this lemma and having length at most $\lfloor 2d\delta^2(n+1)\rfloor$. \\

\noindent
with $\min_{1\le i \le k}|x_n(i)| >0$. We define the nonzero coordinates of $x_{n+1}$ and their number as 
\begin{align}
  J=\{j\in[d]\colon x_{n+1}(j)\neq 0\}\text{ and } l=|J|.
\end{align}
Let $x_n(p)$ and $x_{n+1}(q)$ be the coordinates with the maximal modulus. Then, by \eqref{eq: xn coordinate large}, we have
\begin{equation}
  \label{eq:stayinface}
\begin{split}
  |x_n(p)|\ge 3\delta^2 (n+1),\quad &\max_{1\le i\le d, i\neq p}|x_n(i)|\le (1-3\delta^2)(n+1),\\
  |x_{n+1}(q)|\ge \delta^2 (n+1),\quad &\max_{1\le j\le d, j\neq q}|x_{n+1}(j)|\le (1-\delta^2)(n+1).
\end{split}
\end{equation}
In this case, for $1\le i\le d$ with $i\neq p$, we construct a path $\gamma_n^{i,+}$ from $x_n$ by increasing the $i$-th coordinate and decreasing the $p$-th coordinate as follows:
\begin{align*}
    \gamma_n^{(i,+)}:=(x_n+\bfe_i,x_n+\bfe_i-\bfe_p, \ldots, x_n+m\bfe_i-(m-1)\bfe_p, x_n+m\bfe_i-m\bfe_p,\ldots).
\end{align*}  
By~\eqref{eq:stayinface}, this can be continued up to $m=\delta^2 (n+1)$. For $k+1\le i\le d$, we construct another path $\gamma_n^{i,-}$ by decreasing the $i$-th coordinate and decreasing the \footnote{Modulus?} $p$-th coordinate alternately. Since these paths grow in different directions, they are disjoint and satisfy~\eqref{eq:ps1}--\eqref{eq:ps3} from each other. 
Similarly, for $j\in J$ with $j\neq q$, we construct a path $\gamma_{n+1}^j$ from $x_{n+1}$ by decreasing the modulus of the $q$-th coordinate and increasing the modulus of the $j$-th coordinate alternately. This gives us $l-1$ disjoint paths of length at least $\delta^2 (n+1)$ from $x_{n+1}$ whose endpoints are at least $\delta^2(n+1)$ away from each other. If no such $j\in J$ exists, then $x_{n+1}=(n+1)\bfe_q$ and we must have $q=r$. In this case, we construct a path in the same way as above by choosing any $j$ different from $q$; in all cases we choose distinct prescribed first steps so that initial edges are edge-disjoint as well.

Note that along the above constructions, the $r$-th coordinates of the paths starting from $x_n$ and $x_{n+1}$ always satisfy~\eqref{eq:OppositeSign}. This ensures that the whole collection of paths is disjoint. Moreover, if $|x_n(r)| \ge 2\delta^2 (n+1)$, then by comparing the $r$-th coordinates, we find that
\begin{equation}
\label{eq:EndpointSep2}
\begin{split}
&\text{the endpoints of the paths starting from $x_n$ are at least}\\
&\text{$\delta^2 (n+1)$ away from the union of the paths starting from $x_{n+1}$.}    
\end{split}    
\end{equation}
If $|x_n(r)| < 2\delta^2 (n+1)$ on the other hand, then we know $p\neq r$ from~\eqref{eq:stayinface} and we extend the paths $\gamma_n^{(i,\pm)}$ by adding $\bfe_r$ and $-\bfe_p$ alternately $\delta^2 (n+1)$ times. Thanks to~\eqref{eq:stayinface}, the paths stay in the same orthant as $x_n$ and satisfy~\eqref{eq:EndpointSep2}. In either case, we get the paths from $x_n$ and $x_{n+1}$ which are disjoint and satisfy the requirement in Lemma~\ref{lem:2d-2}.\\

Henceforth, we consider the case that $x_n$ and $x_{n+1}$ are in the same orthant. By applying reflections, we assume that both $x_n$ and $x_{n+1}$ have nonnegative coordinates.
